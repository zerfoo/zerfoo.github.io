Build directly in Go

Feels like Go.
Because it is.

Start with a local GGUF file or download a model from Hugging Face. Load it, ask a question, and build from there.

Follow the quickstart ↗

Go 1.26+ · Model download and available memory depend on your chosen model.

main.go
package main

import (
    "fmt"
    "log"
    "github.com/zerfoo/zerfoo"
)

func main() {
    model, err := zerfoo.Load("./model.gguf")
    if err != nil { log.Fatal(err) }
    defer model.Close()

    reply, err := model.Chat("Why Go?")
    if err != nil { log.Fatal(err) }
    fmt.Println(reply)
}
terminal
# Install the command-line tool
go install github.com/zerfoo/zerfoo/cmd/zerfoo@latest

# Run a local, compatible GGUF model
zerfoo run ./model.gguf

# Serve it over HTTP
zerfoo serve ./model.gguf --port 8080
terminal · start the server using the CLI tab
curl http://localhost:8080/v1/chat/completions \
  -H 'Content-Type: application/json' \
  -d '{
    "model": "./model.gguf",
    "messages": [
      {"role": "user", "content": "Why Go?"}
    ],
    "stream": true
  }'
